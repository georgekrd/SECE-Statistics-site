def boot_mean_ci(values: pd.Series, n_boot: int = 2000, seed: int = 0) -> tuple[float, float, float]:
    """Mean with a 95% percentile-bootstrap interval (papers resampled with replacement)."""
    x = values.dropna().to_numpy(dtype=float)
    idx = np.random.default_rng(seed).integers(0, len(x), size=(n_boot, len(x)))
    lo, hi = np.quantile(x[idx].mean(axis=1), [0.025, 0.975])
    return float(x.mean()), float(lo), float(hi)

def ci_text(group: str, label: bool = True) -> str:
    """Mean score with its interval: '1.20 (95% CI 1.08–1.35)', or '1.20 (1.08–1.35)'."""
    r = G.loc[group]
    ci = f"{dp.fmt_num(r['lo'], 2)}–{dp.fmt_num(r['hi'], 2)}"
    return f"{dp.fmt_num(r['mean'], 2)} ({'95% CI ' if label else ''}{ci})"


def vs_average(group: str) -> str:
    """Mean score relative to the School average (1), e.g. '20% above'."""
    diff = G.loc[group, "mean"] - 1
    return f"{dp.fmt_pct(abs(diff), 0)} {'above' if diff >= 0 else 'below'}"


def vs_interval(group: str) -> str:
    """Position of the 95% interval relative to the School average."""
    r = G.loc[group]
    if r["lo"] > 1:
        return "above the average"
    if r["hi"] < 1:
        return "below the average"
    return "an interval that includes the average, so there is no clear difference"

def rests_on(eids: frozenset) -> int:
    """Distinct SECE authors whose own second affiliations abroad make these papers international."""
    return len(set().union(*held_by.reindex(sorted(eids)).dropna()))


def remainders_safe(cells: list[frozenset]) -> bool:
    """Does every number a reader can form from these cells rest on no or at least MIN_HOLDERS SECE authors? The
    cells must be nested or disjoint; then each such number is a sum of remainders (a cell minus the cells inside
    it), and it is enough to check the remainders."""
    assert all(a <= b or b <= a or not a & b for a in cells for b in cells), "cells must be nested or disjoint"
    rests = [rests_on(c.difference(*[o for o in cells if o < c])) for c in cells]
    return all(n == 0 or n >= MIN_HOLDERS for n in rests)

# Mutually adjusted differences: least squares of the score on all characteristics at once
# (reference: single-institution, closed, no funding acknowledged, no industry partner); both models use
# the same bootstrap resamples of the papers
y = ncs["score"].to_numpy(dtype=float)
idx = np.random.default_rng(0).integers(0, n_ncs, size=(1000, n_ncs))


def adjusted(terms: dict[str, pd.Series]) -> pd.DataFrame:
    """Least-squares differences in mean score for indicator terms, with 95% bootstrap intervals."""
    X = np.column_stack([np.ones(n_ncs)] + [m.to_numpy(dtype=float) for m in terms.values()])
    coef = np.linalg.lstsq(X, y, rcond=None)[0]
    boot = np.array([np.linalg.lstsq(X[i], y[i], rcond=None)[0] for i in idx])
    return pd.DataFrame({"diff": coef[1:], "lo": np.quantile(boot[:, 1:], 0.025, axis=0),
                         "hi": np.quantile(boot[:, 1:], 0.975, axis=0)}, index=list(terms))

         "Open access": ncs["oa_any"], "Funding acknowledged": ncs["has_funding"],
         "Industry partner": ncs["has_industry_partner"], "SECE first or last author": led}
adj = adjusted(TERMS)
# the same model with the partner indicators on the external-partner basis
adj_ext = adjusted({**TERMS, "International": intl_ext, "National": ncs["collab_type_external"] == "national",
                    "Industry partner": ncs["has_industry_partner_external"]})


def adj_text(term: str, table: pd.DataFrame | None = None) -> str:
    """Adjusted difference in mean score with its interval, e.g. '+0.22 (95% CI +0.06 to +0.40)'
    (from `adj`, or from `table`)."""
    r = (adj if table is None else table).loc[term]
    f = lambda v: ("+" if v >= 0 else "−") + dp.fmt_num(abs(v), 2)
    return f"{f(r['diff'])} (95% CI {f(r['lo'])} to {f(r['hi'])})"


def adj_word(term: str, table: pd.DataFrame | None = None) -> str:
    """'positive' / 'negative' / 'uncertain' from the adjusted 95% interval (of `adj`, or of `table`)."""
    r = (adj if table is None else table).loc[term]
    return "positive" if r["lo"] > 0 else "negative" if r["hi"] < 0 else "uncertain"

Key findings

  • By 2 November 2026 the 5,226 publications had received 118,090 citations, a mean of 22.6 and a median of 6 per paper. The School’s h-index is 131 over all years and 88 for the papers published 2016–2025.

  • Citations are concentrated: the most-cited 1% of the papers hold 24.7% of all citations and the most-cited 10% hold 61.7% (all records, of mixed ages; within each publication year from 2000 to 2023 the most-cited 10% hold a median 57.8% of that year’s citations), while 10.8% of the papers published up to 2023 have not been cited.

  • Recent papers have had less time to be cited: papers from 2016–2020 average 29.0 citations, papers from 2021–2025 13.7, and 37.0% of the 2025 papers are still uncited.

  • Compared with papers of the same year and type (2000–2023), internationally co-authored papers are cited 21% above the School average, mainly those led by partners: the mean normalised score is 1.03 (95% CI 0.94–1.13) for international papers with an SECE first or last author and 1.56 (1.24–1.95) for those where SECE holds neither position. Counting only affiliations of co-authors who are not SECE authors, international papers score 1.22 (95% CI 1.09–1.36), so the association holds. Papers with a funding acknowledgement are cited 16% above the average; both differences remain when these characteristics and SECE leadership are considered together.

  • Journal papers are cited far more than conference papers: relative to all articles, reviews and conference papers of the same year (2000–2023), they score 1.58 against 0.29, a factor of 5.5.

Normalised impact by paper characteristics¶

Raw counts mix the effects of age and document type. The normalised citation score divides the citations of each paper by the mean citations of the School’s papers of the same publication year and type: a score of 1 is the School average for comparable papers and 1.5 means 50% more citations than that average. Conference papers are compared with conference papers; articles and reviews are compared together, because most years have too few reviews (59 in these years) for a yearly average of their own. The comparison uses the 3,944 articles, reviews and conference papers published 2000–2023. Earlier years are left out because some of them have fewer than 10 articles and reviews or fewer than 10 conference papers, later ones because their papers have had too little time to be cited; the 99 items of other types in these years (book chapters, editorials, errata and others) are also left out.

Figure 4 shows associations, not causal effects:

  • Collaboration. Internationally co-authored papers score 1.21 (95% CI 1.08–1.34), single-institution papers 0.92 (0.83–1.02) and papers whose only partners are other Greek organisations 0.85 (0.76–0.95). An SECE author’s own second affiliation abroad (a joint, adjunct or visiting position) also makes a paper international. Counting only affiliations of co-authors who are not SECE authors (the external-partner basis, ‡ in the figure), almost all of the international papers remain international; they score 1.22 (95% CI 1.09–1.36), so the association holds. The papers that are international only through such a second affiliation score 0.96 (0.64–1.36), an interval that includes the average.

  • Leadership. International papers with an SECE first or last author score 1.03 (95% CI 0.94–1.13), those led by partners (439 papers, 33.0% of the international ones) 1.56 (1.24–1.95). Their medians differ much less (0.54 and 0.64), so most of the difference comes from a tail of highly cited papers led by partners. On the external-partner basis, partner-led international papers again score higher than SECE-led ones; these two scores are not shown, because with the other values they would reveal a small group of papers (see the notes). Over all papers, those with an SECE first or last author score 0.94 (95% CI 0.88–1.00), the others 1.31 (1.08–1.58).

  • Open access. Open-access papers score 1.08 (95% CI 0.98–1.19) against 0.98 (95% CI 0.91–1.06) for closed ones, but the routes differ: green-only open access (free only as a repository copy) scores 1.40 (95% CI 1.21–1.58), while gold open access (papers in fully open-access journals) scores 0.82 (95% CI 0.69–0.96), below the average.

  • Funding. Papers that acknowledge funding score 1.16 (95% CI 1.05–1.30), papers without an acknowledgement 0.92 (95% CI 0.85–1.00).

  • Industry. The 117 papers with an industry partner score 0.97 (95% CI 0.75–1.21), an interval that includes the average, so there is no clear difference.

  • Source type. Relative to all articles, reviews and conference papers of the same year, journal papers score 1.58 (95% CI 1.44–1.74) and conference papers 0.29 (95% CI 0.26–0.32), a factor of 5.5.

These characteristics overlap: 45.8% of the international papers acknowledge funding and 36.9% are open access, against 27.0% and 12.7% of the other papers. A least-squares model of the score on collaboration type, open access, funding, industry partnership and SECE leadership (an SECE first or last author) together (bootstrap 95% intervals) estimates each difference with the others held fixed. Compared with single-institution papers, international co-authorship is associated with a difference in mean score of +0.19 (95% CI +0.04 to +0.34) and national co-authorship with −0.11 (95% CI −0.25 to +0.03). A funding acknowledgement is associated with +0.19 (95% CI +0.04 to +0.35), open access with −0.06 (95% CI −0.23 to +0.09), an industry partner with −0.21 (95% CI −0.49 to +0.05) and an SECE first or last author with −0.28 (95% CI −0.55 to −0.05). With the other characteristics held fixed, the international difference is positive, the funding difference positive, the open-access difference uncertain and the difference for an SECE first or last author negative. With the international, national and industry-partner indicators on the external-partner basis, the international difference is +0.21 (95% CI +0.05 to +0.35), also positive. Research area and venue, which the score does not normalise, can still explain part of all these differences.

Notes & caveats¶

  • Snapshot. Citation counts are the Scopus “Cited by” values on 2 November 2026. They include self-citations, count only citing documents indexed in Scopus, and keep growing; other databases (Web of Science, Google Scholar) give different counts.

  • Partial year. 2026 covers records up to 2 November 2026 (246 papers). They are included in the all-records statistics (totals, distribution, concentration), but not in the citations by publication year (Figure 3), the windowed h-indices, the period averages or the normalised scores.

  • School average, not world average. The normalised score compares SECE papers with other SECE papers of the same year and type (articles and reviews together, conference papers apart). It is not a field-normalised indicator such as the field-weighted citation impact (FWCI) and says nothing about impact relative to the world. Research areas with different citation habits are not separated. Because the score averages 1 in every year by construction, it cannot show whether impact rose or fell over time.

  • Skewed data. Means depend on a few highly cited papers; the bootstrap intervals show this uncertainty. As a check, the group medians rank the groups as the means do for collaboration type, open access vs closed, funding, source type, SECE first or last author and leadership of the international papers, but not for industry partner (medians in the hover of Figure 4).

  • Flags. Open-access flags describe the status at the export date, not at publication. Funding acknowledgements in Scopus are incomplete, especially for older papers, so some papers counted without funding were funded. Industry partners are found through affiliations matched to ROR organisations of type company. 89.1% of the affiliation occurrences outside AUTH are matched to a ROR organisation, and many companies have no ROR record, so this group is undercounted. SECE first and last authors are identified from each author’s affiliation text (see Research community), and author order is an imperfect signal of who led a paper.

  • Definitions. “Type” is the Scopus document type. The source type (journal, conference, book) is derived from the document type, not from the venue, so a conference paper published in a journal issue counts as conference, as on Research output. International papers have at least one affiliation in a country other than Greece, read from the affiliation addresses as SciVal and the Leiden Ranking do; the external-partner variant (‡ in Figure 4) counts only affiliations of co-authors who are not SECE authors. Collaboration types, open-access categories and the other definitions are described in Methodology & data.

  • Small groups. A number of papers that are international only through SECE authors’ own second affiliations abroad, or a score of such papers, is shown only when at least 3 SECE authors hold these affiliations, since with fewer it would in practice describe one or two people. This also covers numbers that could be derived from the values shown, on this page or together with those of other pages, such as the difference between the address-based and the external-partner count of the same group. The site gives the number of these papers only for the years, periods and spans of years chosen for all its pages together (see Methodology & data); the articles, reviews and conference papers of 2000–2023 compared here are not among them, so the external-partner groups are shown without their number of papers.
