def normalize_title(title: str) -> str:
    """NFKC, plain hyphens, no bracketed translations, lower case."""
    text = unicodedata.normalize("NFKC", title).translate(DASHES)
    return BRACKETS.sub(" ", text).replace("’", "'").lower()


def singular(word: str) -> str:
    """Conservative English singular (networks -> network, technologies -> technology)."""
    if word in IRREGULAR:
        return IRREGULAR[word]
    if len(word) <= 3 or word in KEEP_S or not word.endswith("s") or word.endswith(("ss", "us", "is", "ics")):
        return word
    if word.endswith("ies") and len(word) > 4:
        return word[:-3] + "y"
    if word.endswith(("sses", "ches", "shes", "xes")):
        return word[:-2]
    return word[:-1]


def tokenize(title: str) -> tuple[list[str], list[str]]:
    """(normalised tokens, surface forms) of a title; hyphenated words are split."""
    text = normalize_title(title)
    for rx, tok in PHRASES:
        text = rx.sub(tok, text)
    raw = [m.group(0).replace("'s", "").replace("'", "") for m in WORD.finditer(text)]
    return [singular(w) for w in raw], raw


def is_content(word: str) -> bool:
    return len(word) > 1 and word not in STOP and not re.fullmatch(r"\d+(?:\.\d+)?", word)


def title_terms(title: str) -> set[str]:
    """Unigrams (content words that are not generic modifiers) and bigrams of adjacent content words."""
    toks, _ = tokenize(title)
    keep = [is_content(w) for w in toks]
    uni = {w for w, k in zip(toks, keep) if k and w not in WEAK}
    bi = {f"{a} {b}" for a, ka, b, kb in zip(toks, keep, toks[1:], keep[1:]) if ka and kb and a != b}
    return uni | bi

def top_terms(mask, total: int, n: int = N_TOP) -> pd.DataFrame:
    """The ``n`` most frequent terms plus any terms tied with the n-th (ties in alphabetical order)."""
    counts = doc_freq(mask)

    out["share"] = out.titles / total
    out["label"] = out.term.map(show)
    return out

def npmi(top: np.ndarray) -> float:
    vals = []
    for i, a in enumerate(top):
        for b in top[i + 1:]:
            p_ab = B[:, a].multiply(B[:, b]).sum() / n_titles
            vals.append(-1.0 if p_ab == 0 else np.log(p_ab / (p_term[a] * p_term[b])) / -np.log(p_ab))
    return float(np.mean(vals))

# ------------------------------------------------------------------ rising and declining terms
# Weighted log-odds ratio with an informative Dirichlet prior (Monroe, Colaresi & Quinn 2008):
# counts = titles containing the term in P_RECENT vs P_BASE; prior = counts in all titles of the full years.
y_r, y_b, prior = doc_freq(is_recent), doc_freq(is_base), doc_freq(in_full)

n_r, n_b = lo.recent.sum(), lo.base.sum()   # term occurrences (one per title and term) per period


def log_odds_z(scale: float = 1.0) -> pd.Series:
    """z-score of the log-odds difference, prior = ``scale`` x the term counts of all full years."""
    alpha = scale * prior.reindex(lo.index).astype("float64")
    a0 = scale * float(prior.sum())
    delta = (np.log((lo.recent + alpha) / (n_r + a0 - lo.recent - alpha))
             - np.log((lo.base + alpha) / (n_b + a0 - lo.base - alpha)))
    return delta / np.sqrt(1 / (lo.recent + alpha) + 1 / (lo.base + alpha))


lo["z"] = log_odds_z(1.0)
lo = lo[~lo.index.isin(SUBSUMED)].copy()
lo["share_recent"], lo["share_base"] = lo.recent / n_recent, lo.base / n_base

Most frequent terms¶

Each title is reduced to its terms. It is lower-cased, punctuation is removed, plural forms are merged (networks → network), and spelled-out and abbreviated forms count together (finite-difference time-domain → FDTD, internet of things → IoT, non-orthogonal multiple access → NOMA). English stop words and generic research words (using, based, approach, analysis, novel, new, study, method, application, towards, via, …) are dropped. Generic modifiers such as system, model, design, time, power, high and low count only inside two-word terms (power system, real time, low voltage). A term is a single word or a pair of adjacent words, and a word that occurs mostly inside one pair (neural → neural network) is shown as that pair. A term counts once per title; the titles have a median length of 11 words.

Research topics¶

A topic model finds groups of terms that tend to occur in the same titles. Each title is represented by its terms, weighted by TF-IDF so that terms common to the whole collection count less, and non-negative matrix factorisation (NMF) describes every title as a weighted mix of 12 topics, each of which is a weighted list of terms. The 2,316 terms that occur in at least 5 titles enter the model. Each topic was named after reading its top terms and the titles closest to it, and the topics are grouped into 3 broad areas (Energy & power; Communications & electromagnetics; Computing, signals & control). Every title is counted under its dominant topic, the one with the largest weight.

Citation impact by topic¶

The normalised citation score divides the citations of a paper by the mean citations of the department’s papers of the same publication year and type (articles and reviews together), as on Citation impact: a score of 1 is the department average. Wireless & optical communications scores highest (Figure 3), 1.55 (95% CI 1.20–2.00), and Computational EM & photonics lowest, 0.53 (95% CI 0.45–0.62). The intervals of 2 topics (Wireless & optical communications and Fading channels & diversity) lie entirely above 1 and those of 2 (Computational EM & photonics and Lightning & high voltage) entirely below; the others include the average. These are associations, not effects of the topic itself: research fields differ in how much and how fast they cite, and the topics differ in venues, collaboration and open access, so a topic’s score mixes the department’s standing in the field with the citation habits of that field.

Rising and declining terms¶

Which terms became more or less common? The titles of 2011–2015 (877) are compared with those of 2021–2025 (1,282) by the weighted log-odds ratio with an informative Dirichlet prior (Monroe, Colaresi & Quinn, 2008). For each term it takes the difference between the two periods in the log-odds of the term appearing, shrinks it towards the term’s frequency in all titles of 1981–2025 (the prior) and divides it by its standard error. The resulting z-score is large only for terms that are both frequent and unevenly spread, so rare terms do not dominate the list.

Notes & caveats¶

  • Titles only. The topics are read from the titles alone: every record has a title, so the same method covers all years, and the processed data the site is built from carry no abstracts or keywords. Titles are short and often name a method rather than the application (or the reverse), so term counts and topics describe emphasis, not an exact classification: a title about a neural-network method for antenna design counts for both terms.

  • Text preparation. The stop words are scikit-learn’s English list without the words that are engineering terms (system, side, thin, full, …) plus generic research words. Plural forms are merged by simple rules, spelled-out phrases are folded into their abbreviations, and translated titles in brackets are removed. These rules are approximate.

  • Topic model. TF-IDF of the single words and word pairs that occur in at least 5 titles (2,316 terms) and NMF from scikit-learn (initialisation nndsvda, random_state = 0), which gives the same result on every run for a given scikit-learn version and text preparation. Models with 10 to 15 topics were compared by topic coherence, the mean normalised pointwise mutual information (NPMI) of the ten top terms of each topic, from their co-occurrence in the titles. Coherence was highest with 12 topics (0.138) and decreased with fewer or more topics, to 0.059 with 10 and 0.052 with 15; the page uses 12 topics, the most coherent choice. Coherence and the topics themselves also shift when the text preparation changes slightly, so the choice also rests on inspection: with 12 topics the department’s main fields appear as separate, interpretable topics. Refitted on 10 random samples of 80% of the titles, the matching topics share on average 8.0 of their ten top terms with the full model, from 4.0 for Distribution grids & renewables to 9.6 for Control & robotics.

  • Topic names and assignment. The names were chosen by hand and are matched to the model’s topics through anchor terms; all 12 topics received a name. Topics are broader than their names: health-related titles, for example, count most often under Machine learning and edge-computing titles under Wireless & optical communications, and most titles that mention artificial intelligence count under Antennas & metamaterials, the topic in which the model gives that term its largest weight. Between 45.4% (Computational EM & photonics) and 86.4% (Energy management & storage) of a topic’s titles contain at least one of its 8 top terms; the others are assigned through further terms that the model links to the topic. Each title counts under its dominant topic, although for 34.4% of the titles the dominant topic carries less than half of the title’s topic weight. Fractional topic weights give similar changes between 2011–2015 and 2021–2025 (correlation 0.98 across topics). The three largest losses are then in Computational EM & photonics, Fading channels & diversity and Image & signal analysis.

  • Citation scores. As on Citation impact, the counts are Scopus “Cited by” values on 2 November 2026 (self-citations included), and the scores compare DECE papers with other DECE papers of the same year and type (articles and reviews together), not with the world. Means depend on a few highly cited papers (in every topic the median score is below the mean); the 2,000 bootstrap resamples show this uncertainty, and small topics tend to have wide intervals.

  • Rising and declining terms. The z-scores describe the difference between two periods; they are not significance tests and are not corrected for the 8,941 terms compared. With a prior of a quarter of the strength, 14 of the 15 rising and 12 of the 15 declining terms stay the same. Several declining terms belong to one line of work, silicon plasmonics (plasmonic, silicon, dielectric loaded, silicon plasmonic and plasmonic waveguides).

  • Selected technologies. Mentions are found in titles with regular expressions, and the abbreviations EV and PV are taken to mean electric vehicle and photovoltaic. The 3-year windows smooth year-to-year noise, but the early windows rest on few titles (76 in 1993–1995). The partial year 2026 is left out of all trends and periods.

  • Coverage. The publications are those of the Scopus affiliation query described in Methodology & data.
