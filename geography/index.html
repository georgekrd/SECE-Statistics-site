def region_of(c: pd.Series) -> str:
    """World region of a country: EU-27 membership first, then continent / UN subregion."""
    if c["is_eu27"]:
        return "EU-27"
    if c["continent"] == "Europe":
        return "Other Europe"
    if c["continent"] == "North America":
        return "North America"
    if c["subregion"] == "Western Asia":
        return "Western Asia"
    if c["continent"] == "Asia":
        return "Rest of Asia"
    return "Rest of world"


def prose(code: str, start: bool = False) -> str:
    """Country name for running text ("the United Kingdom"), capitalised at a sentence start."""
    name = countries.at[code, "country_name"]
    if name.startswith("United ") or name in {"Netherlands", "Philippines"}:
        name = f"the {name}"
    return name[:1].upper() + name[1:] if start else name


def and_join(names: list[str]) -> str:
    """'a', 'a and b', 'a, b and c'."""
    return names[0] if len(names) == 1 else ", ".join(names[:-1]) + " and " + names[-1]


def and_list(codes: list[str]) -> str:
    """'India and South Korea' from country codes."""
    return and_join([prose(c) for c in codes])


def ordinal(rank: int) -> str:
    """'second' for rank 2 (ranks past 10 as 'number 11')."""
    words = ["first", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth", "ninth", "tenth"]
    return words[rank - 1] if 1 <= rank <= len(words) else f"number {rank}"


def sole_max(s: pd.Series):
    """The index of the largest value of *s*, or None where several share it (for the guards of rankings)."""
    top = s.index[s.eq(s.max())]
    return top[0] if len(top) == 1 else None


def nth(n: int) -> str:
    """'20th', '21st', '22nd', '23rd'."""
    return f"{n}{'th' if 10 <= n % 100 <= 20 else {1: 'st', 2: 'nd', 3: 'rd'}.get(n % 10, 'th')}"

def held_by_year(dece_only: pd.DataFrame, key: str) -> dict:
    """For each *key* value (a country, region or ror_id_top) of *dece_only*, the (paper, key) pairs
    linked only through DECE authors' own second affiliations: {year: the DECE authors who hold them}."""
    held = own_holders.merge(dece_only[["eid", key, "year"]].drop_duplicates(), on=["eid", key])
    return {k: g.groupby("year")["author_id"].agg(set).to_dict() for k, g in held.groupby(key)}


def rests_on(held: dict, years: frozenset) -> int:
    """Distinct DECE authors behind a place's links of these years (held: the place's held_by_year)."""
    return len(set().union(*(held.get(year, set()) for year in years)))


def shown_cells(held: dict, cells: dict[str, frozenset]) -> dict[str, bool]:
    """Which external-partner values of a place may be shown. *cells* maps each value that the page gives
    for the place (all years, a period, ...) to its years, in order of importance; *held* is the place's
    held_by_year. Next to the count of all affiliations, an external-partner value gives the number of
    papers linked to the place only through DECE authors' own second affiliations, and two values give
    the difference of these numbers (all years minus a period, say). A value is shown when, with the
    values shown before it, every number that follows rests on no DECE author or on at least MIN_HOLDERS.
    The cells are nested or disjoint, so these numbers are sums of remainders (a cell minus the cells
    shown inside it), and it is enough to check the remainders."""
    shown, out = [], {}
    for name, years in cells.items():
        trial = [*shown, years]
        assert all(a <= b or b <= a or not a & b for a in trial for b in trial), "cells must be nested or disjoint"
        rests = [rests_on(held, cell.difference(*[c for c in trial if c < cell])) for cell in trial]
        out[name] = all(n == 0 or n >= MIN_HOLDERS for n in rests)
        if out[name]:
            shown.append(years)
    return out

    return f"{'most' if 2 * m > n else 'half' if 2 * m == n else 'part'} of the {dp.fmt_int(n)}"

def main_inst(code: str) -> str:
    """Map hover: a country's most frequent partner institution and its papers; where several tie, all of them

    return "<br>".join(textwrap.wrap(f"{and_join(names)} {tied}", 60, break_on_hyphens=False, break_long_words=False))

def km(lat1: float, lng1: float, lat2: float, lng2: float) -> float:
    """Great-circle distance in kilometres between two points given in degrees."""
    lat1, lng1, lat2, lng2 = np.radians([lat1, lng1, lat2, lng2])
    cos = np.sin(lat1) * np.sin(lat2) + np.cos(lat1) * np.cos(lat2) * np.cos(lng2 - lng1)
    return float(6371 * np.arccos(min(1.0, cos)))

def unit_space(values: dict) -> tuple:
    """The values as a matrix over the units (merged where no value tells them apart), the papers of the units
    (papers x units) and the sets of units that rest on fewer than MIN_HOLDERS DECE authors, one for each
    group of such authors."""
    unit = sorted(set().union(*values.values()), key=lambda u: (u[0], sorted(u[1])))
    if not unit:                                  # no paper counts through DECE authors' own affiliations
        return np.zeros((len(values), 0)), np.zeros((0, 0)), []
    F = np.array([[val.get(u, 0) for u in unit] for val in values.values()], dtype=float)
    _, first, cls = np.unique(F.T, axis=0, return_index=True, return_inverse=True)
    cls = np.ravel(cls)
    who = [frozenset().union(*(unit[i][1] for i in np.flatnonzero(cls == k))) for k in range(len(first))]
    row_of = {e: i for i, e in enumerate(sorted({u[0] for u in unit}))}
    paper_of = np.zeros((len(row_of), len(first)))
    for i, u in enumerate(unit):
        paper_of[row_of[u[0]], cls[i]] += 1
    people = sorted(set().union(*who))
    small = {tuple(k for k, w in enumerate(who) if w <= set(g))
             for n in range(1, MIN_HOLDERS) for g in combinations(people, n)} - {()}
    return F[:, first], paper_of, sorted(small)


def revealed(Q: np.ndarray, paper_of: np.ndarray, small: list) -> tuple | None:
    """A set of *small* on which some combination of the values (Q: orthonormal basis of the combinations) is a
    number of papers other than zero, or None."""
    P = Q.T @ Q
    for g in small:
        ev, vec = np.linalg.eigh(P[np.ix_(g, g)])
        on_g = vec[:, ev > 1 - 1e-8]              # the combinations that are zero outside g
        if on_g.size and np.abs(paper_of[:, g] @ on_g).max() > 1e-7:
            return g
    return None

Partner institutions abroad¶

The international papers involve 735 partner institutions abroad (top-level ROR organisations) in 501 cities and 76 countries; 95.8% of the international papers have at least one of them (Figure 6). Of these institutions, 730 appear through at least one co-author who is not a DECE author; for 9, at least half of the papers come only through DECE authors’ own second affiliations. Universities dominate: 587 of the 735 institutions belong to the education sector and 39 are companies; the others are mostly research facilities, public bodies (including national research organisations such as the Fraunhofer-Gesellschaft), hospitals and nonprofits. Three institutions appear most often: Khalifa University of Science and Technology (United Arab Emirates, 280 papers), the University of Huddersfield (United Kingdom, 128) and the Lebanese American University (Lebanon, 96). On the external-partner basis, which leaves out the papers that name them only in DECE authors’ own second affiliations, Khalifa University of Science and Technology keeps 187 of its 280 papers, the University of Huddersfield almost all of its 128 and the Lebanese American University fewer than half of its 96; Southwest Jiaotong University is then third. Together, the three appear on 44.1% of the international papers of 2021–2025, up from 27.6% in 2016–2020 and 10.3% in 2011–2015; on the external-partner basis, on 29.4% of the 599 international papers of 2021–2025, also more than in 2016–2020. They also account for part of the rise in multinational papers: without their papers, 33.1% of the international papers of 2021–2025 involve two or more countries abroad, against 48.7% overall and 24.0% in 2001–2005.

Notes & caveats¶

  • Country attribution. Countries come from the affiliation text in Scopus (with the ROR record as a fallback). AUTH affiliations count as Greece, and memberships of societies such as IEEE are ignored. Affiliations outside AUTH without a recognisable country (16, on 12 papers) are not counted. Hong Kong, Macao and Taiwan are counted separately from China, as in ISO 3166.

  • Dual affiliations and external partners. Scopus lists each author’s own affiliations, so every affiliation on a paper can be attributed to the authors who hold it, with the same split of the per-author affiliation text that identifies the DECE authors (see Methodology & data); all 13,025 affiliation occurrences (society memberships left out) have at least one holder. An affiliation outside AUTH whose holders are all DECE authors is a DECE author’s own second affiliation (a joint, adjunct or visiting position): 325 of the 4,541 affiliations abroad on DECE papers are of this kind, on 297 papers. The headline figures count them like any other affiliation, as address-based indicators do; the external-partner values leave them out, but keep an affiliation that a DECE author shares with a co-author who is not a DECE author. Of the 1,953 international papers, 97 are international only through DECE authors’ own second affiliations: 8 of the 104 of 2001–2005, 50 of the 649 of 2021–2025 and 11 of the 169 of 2025. They weigh most on Western Asia: most of the 96 papers with the Lebanese American University (most of the 89 in 2021–2025) and 93 of the 280 with Khalifa University of Science and Technology (part of the 122) name the institution only in DECE authors’ own second affiliations. Shares on the external-partner basis have fewer international papers as denominator, so a country or region whose papers hardly change has a slightly higher share on that basis. Both bases describe institutional links, not where individual authors live or come from.

  • Small groups. The number of papers that involve a country, a region or an institution abroad, or any country abroad at all, only through DECE authors’ own second affiliations is given only when it is zero or at least 3 DECE authors hold these affiliations, and so is any number that follows from the values shown, on this page or together with those of other pages, such as the difference between two of them (the value of all years minus those of the periods, a region minus its countries, the international papers minus a country’s, a period minus its years on the Collaboration page). With fewer, it would in practice be one or two people’s paper count, which this site does not publish: the page then describes it in words and leaves out the external-partner values that would reveal it (“not shown” in the hovers, “–” in Table 1 and in the period labels of Figure 3 and Figure 4, a missing tick in Figure 2 and gaps in the dashed lines of Figure 4), keeping those of 2021–2025 and of all years where it can. The values for all countries together are chosen for the whole site at once: a period without them here also has none in the period table of the Collaboration page.

  • Full counting. A paper with affiliations in several countries counts once for each country and region, so the counts per country add up to more than the 1,953 international papers.

  • Regions. Continents and UN subregions follow Natural Earth. EU-27 is the current membership, applied to all years: the United Kingdom is outside the EU also before 2020, and Cyprus, which Natural Earth places in Western Asia, counts as EU-27.

  • Institutions. Only affiliations matched to a ROR record can be attributed to an institution: 91.4% of the affiliations abroad are matched (see Methodology & data); the others still count for their country. A partner institution abroad is a top-level ROR organisation outside Greece named in an affiliation outside Greece, so a Greek branch of a foreign organisation does not count (1 in these data; the Collaboration page counts partners with the same rule). On the map each matched unit is placed in its own city, as ROR locates it, so a research institute appears in its own city rather than at its parent’s headquarters. ROR locations are city-level: the institutions of one city and sector share a bubble. Sectors are the ROR organisation types, as on the Collaboration page.

  • Maps. Country shapes are loaded from the plotly CDN when the page is viewed; very small countries are not drawn at world scale but are listed under Table 1.

  • Partial year. 2026 covers papers to 2 November 2026. It is included in the totals and on the maps, shown separately in Figure 5, and left out of the comparisons between periods.

  • Interpretation. These are associations in co-authorship data. A few institutions that appear on many papers weigh heavily in the country and region shares; the shares describe the countries and institutions named in the affiliations, not where individual authors are based, nor the value or the formal status of any partnership.
