def and_join(items: list[str]) -> str:
    """'a', 'a and b', 'a, b and c'."""
    items = list(items)
    return items[0] if len(items) == 1 else ", ".join(items[:-1]) + " and " + items[-1]


def fold_text(ratio: float) -> str:
    """A growth by a factor of about two to four in words: 'doubled', 'more than doubled', 'nearly tripled'..."""
    words = {2: "doubled", 3: "tripled", 4: "quadrupled"}
    assert 1.9 <= ratio < 4.1, f"text: no wording for a growth by {ratio:.2f} times"
    k = round(ratio)
    if abs(ratio - k) <= 0.1:
        return words[k]
    if ratio < k <= ratio + 0.25:
        return f"nearly {words[k]}"
    return f"more than {words[int(ratio)]}"


def latest_text(share: pd.Series) -> str:
    """The latest full year against the other full years: 'reached <share> in <LAST_FULL>, its highest of any
    full year', or 'was <share> in <LAST_FULL>, after a high of <share> in <year>' when an earlier full year
    has the highest share."""
    best = share.loc[:LAST_FULL].idxmax()
    if best == LAST_FULL:
        return f"reached {dp.fmt_pct(share[LAST_FULL])} in {LAST_FULL}, its highest of any full year"
    return f"was {dp.fmt_pct(share[LAST_FULL])} in {LAST_FULL}, after a high of {dp.fmt_pct(share[best])} in {best}"


def top_rows(ranking: pd.DataFrame, col: str, n: int) -> pd.DataFrame:
    """The first n rows of a ranking sorted by *col* (largest first) and any further rows tied with the n-th,
    so that the selection does not depend on how ties are broken."""
    return ranking[ranking[col].ge(ranking[col].iloc[n - 1])] if len(ranking) > n else ranking


def separated(ranking: pd.DataFrame, col: str, k: int) -> bool:
    """True when the first k rows of a ranking sorted by *col* (largest first) rank above all the others, so
    that the k named first do not depend on how ties are broken."""
    return len(ranking) <= k or ranking[col].iloc[k - 1] > ranking[col].iloc[k]

def unshown_years(t: str) -> list[int]:
    """The years whose difference for type t follows only together, from all years minus the parts shown: those
    of the periods that hide it and the partial year, without (international) the years the chart splits."""
    years = [y for p in PERIODS if not cell_shown.at[p, t] for y in years_of(p)] + [PARTIAL]
    return [y for y in years if t != "international" or not split_shown.get(y, False)]


def tint(color: str, amount: float = 0.35) -> str:
    """Hex colour mixed with white (0: unchanged, 1: white)."""
    rgb = (int(color[i:i + 2], 16) for i in (1, 3, 5))
    return "#" + "".join(f"{round(c + (255 - c) * amount):02x}" for c in rgb)


def both_pct(addr: pd.Series, ext: pd.Series, shown: pd.Series) -> list[str]:
    """'48.3% (44.7%)': the share counting every affiliation, with the external-partner share in
    brackets where the two differ after rounding, or '48.3% (–)' where it is not shown (small groups)."""
    return [a if ok and a == e else f"{a} ({e if ok else '–'})"
            for a, e, ok in zip(addr.map(dp.fmt_pct), ext.map(dp.fmt_pct), shown)]

def prose(org: str) -> str:
    """Partner name for running text: no "(Country)" suffix, "the" where English needs it."""

def top_text(df: pd.DataFrame, col: str, k: int, unit: str = " papers", own: str | None = None) -> str:
    """'A (n papers), B (n) and C (n)' for the first k rows of a ranking. With *own* (the column of
    papers only through SECE authors' own second affiliations), an institution where these are at least
    half of its papers reads 'A (n, m of them only through ...)', or 'A (n, most of them only through
    ...)' when m may not be shown (column *own*_shown)."""
    parts = []
    for i, row in enumerate(df.head(k).to_dict("records")):
        text = f"{dp.fmt_int(row[col])}{unit if i == 0 else ''}"
        if own is not None and 2 * row[own] >= row[col] > 0:
            m = dp.fmt_int(row[own]) if row[f"{own}_shown"] else most_or_half(row[own], row[col])
            text += f", {m} of them {OWN_TEXT}"
        parts.append(f"{prose(row['ror_id'])} ({text})")
    return and_join(parts)


def own_examples(k: int) -> str:
    """'m of the n papers with A and m of the n with B link to the institution only through ...': the k
    partners with the most such papers among the counts that may be shown, then ', and so do most of
    the n papers with C' for the partners of the all-years figure whose count is not shown (mostly_hidden)."""
    shown = [f"{dp.fmt_int(m)} of the {dp.fmt_int(n)}{' papers' if i == 0 else ''} with {prose(org)}"
             for i, (org, m, n) in enumerate(zip(most_own["ror_id"].head(k), most_own["only"].head(k),
                                                  most_own["papers"].head(k)))]
    hidden = [f"{most_or_half(m, n)} of the {dp.fmt_int(n)} papers with {prose(org)}"
              for org, m, n in zip(mostly_hidden["ror_id"], mostly_hidden["only"], mostly_hidden["papers"])]
    return (f"{and_join(shown)} link to the institution {OWN_TEXT}"
            + (f", and so do {and_join(hidden)}" if hidden else ""))

def partner_bars(df: pd.DataFrame, value: str, own: str, value_title: str,
                 other: str, other_own: str, other_title: str) -> go.Figure:
    """Horizontal bars of partner institutions (largest on top), coloured by sector.

    The hatched end of a bar counts the papers that link to the institution only through SECE
    authors' own second affiliations (column *own*), drawn only where that count may be shown
    (column *own*_shown); an asterisk marks the institutions where these are at least half of the
    papers. *other* / *other_own*: the same counts for the other period (hover).
    """
    rows = df.iloc[::-1].reset_index(drop=True)   # plotly draws the first category at the bottom
    star = np.where(2 * rows[own] >= rows[value], "*", "")
    rows["label"] = [dp.wrap_label(f"{n}{s} · {c}", 46) for n, s, c in zip(rows["name"], star, rows["country_name"])]
    rows["place"] = [f"{city}, {country}" if isinstance(city, str) else country
                     for city, country in zip(rows["city"], rows["country_name"])]
    # hover values as text, so that a count that may not be shown reads "not shown"
    for col in (value, other):
        rows[f"{col}_text"] = rows[col].map(dp.fmt_int)
    not_shown = "not shown (small groups)"
    for col in (own, other_own):
        rows[f"{col}_text"] = [dp.fmt_int(n) if ok else not_shown for n, ok in zip(rows[col], rows[f"{col}_shown"])]
    custom = rows[["name", "place", "sector", f"{value}_text", f"{own}_text", f"{other}_text", f"{other_own}_text",
                   "first_year"]].to_numpy()
    hover = ("<b>%{customdata[0]}</b><br>%{customdata[1]}<br>%{customdata[2]}"
             f"<br>{value_title}: %{{customdata[3]}}<br>of which {OWN_TEXT}: %{{customdata[4]}}"
             f"<br>{other_title}: %{{customdata[5]}}<br>of which {OWN_TEXT}: %{{customdata[6]}}"
             "<br>First joint paper: %{customdata[7]}<extra></extra>")
    fig = go.Figure()
    for sector, color in SECTOR_COLORS.items():
        part = rows["sector"].eq(sector).to_numpy()
        if not part.any():
            continue
        fig.add_bar(x=rows.loc[part, value], y=rows.loc[part, "label"], orientation="h", name=sector,
                    marker_color=color, customdata=custom[part], hovertemplate=hover,
                    texttemplate="%{x:,}", textposition="outside")
    # papers only through SECE authors' own second affiliations: a lighter, hatched end drawn over the
    # bar, where the count may be shown
    part = (rows[own].gt(0) & rows[f"{own}_shown"]).to_numpy()
    fig.add_bar(x=rows.loc[part, own], base=(rows[value] - rows[own])[part], y=rows.loc[part, "label"],
                orientation="h", name=OWN_LABEL, legendgroup="own", showlegend=False,
                marker=dict(color="rgba(255,255,255,0.35)", pattern=OWN_PATTERN),
                customdata=custom[part], hovertemplate=hover)
    fig.add_bar(x=[None], y=[None], orientation="h", name=OWN_LABEL, legendgroup="own", hoverinfo="skip",
                marker=dict(color=dp.OTHER_COLOR, pattern=OWN_PATTERN))   # legend key only
    fig.update_layout(height=dp.fig_height(len(rows), per_row=30), barmode="overlay", bargap=0.3)
    fig.update_xaxes(title_text=value_title, showgrid=True, showline=False, zeroline=False,
                     range=[0, float(rows[value].max()) * 1.15])
    fig.update_yaxes(categoryorder="array", categoryarray=list(rows["label"]), showgrid=False,
                     showline=True)
    return fig

def vague(rows: pd.DataFrame, col: str, own: str, text: bool = True) -> bool:
    """Counts not shown that the text calls 'most' (never 'half', which would give the number) or that an

Partner institutions¶

A partner institution is the top-level ROR organisation behind an affiliation outside AUTH: a university, research centre, hospital, company or public body. Institutes and hospitals that ROR places under a parent organisation count under the parent; universities count on their own, even within a university system. Since 1981, SECE papers have involved 810 distinct partner institutions, 74 of them in Greece and 735 abroad; 1 foreign institution appears only through a campus or branch in Greece. Counting only affiliations of co-authors who are not SECE authors, 805 remain (730 abroad); the other 5 appear only through SECE authors’ own second affiliations.

Greek partner landscape¶

The 74 Greek partner institutions have changed places over time (Figure 6). CERTH with its ITI was the main Greek partner in the 2000s and peaked in 2006–2010 (104 papers, or 75 counting only affiliations of co-authors who are not SECE authors); in 2021–2025 it appeared on 51, nearly all of them with co-authors from CERTH or ITI who are not SECE authors, so part of the decline comes from fewer papers on which CERTH or ITI appears only through SECE authors’ own second affiliations. The most frequent Greek partners of 2021–2025 were the University of Western Macedonia (109 papers, up from 32 in 2011–2015) and the Democritus University of Thrace (67, up from 11). Local partners weigh heavily: institutions based in Thessaloniki appear on 37.6% of the 1,562 papers with a matched Greek partner institution. Part of the change is administrative: the technological educational institutes (TEIs) were merged into universities in 2018–2019, so, for example, papers with the Alexander TEI of Thessaloniki give way to papers with the International Hellenic University.

Breadth of the partner network¶

The partner network has widened (Figure 7). Papers of 2021–2025 involved 140.0 distinct partner institutions a year on average, 4.6 times the 30.2 of 2001–2005. Partners per paper¶

Papers increasingly involve several partner institutions (Figure 8). Papers with two or more rose from 15.2% of the output in 2001–2005 to 37.6% in 2021–2025, and papers without a matched partner fell from 53.7% to 30.5%. The mean number of partner institutions per paper went from 0.69 to 1.50; the most on a single paper is 27, on a paper of 2026.

Notes & caveats¶

  • Affiliations, not people. Collaboration types and partners come from the affiliations listed on each record. A paper is international when at least one affiliation is outside Greece (unmatched strings count through the country they name), national when it has a partner affiliation in Greece and none abroad, and single-institution otherwise; Methodology & data gives the full definitions. This address-based rule is the usual one (SciVal and the Leiden Ranking, for example, count international collaboration from the countries in the affiliations), and it counts every affiliation, including a second affiliation that an SECE author holds abroad or at another Greek institution.

  • SECE authors’ own second affiliations. Scopus also lists the affiliations of each author, so an affiliation outside AUTH counts as an SECE author’s own when every author of the paper who lists it is an SECE author (an author whose own affiliation names the School). There are 534 such affiliations on 475 papers: 325 abroad, 199 in Greece and 10 without a recognisable country; every affiliation on the records is listed by at least one of their authors. Counting only affiliations of co-authors who are not SECE authors, 97 of the 1,953 international papers (5.0%) are not international and 103 of the 1,133 national papers (9.1%) become single-institution (Table 1 gives the shares by period). Most other figures on this page change little: in 2021–2025, for example, 61.2% of the papers have a university partner on that basis (63.7% counting every affiliation), and a paper has 1.37 partner institutions on average (1.50).

  • Small groups. No number of papers that link to an institution, a sector or a collaboration type only through SECE authors’ own second affiliations is shown, in any year or period, unless it is zero or at least 3 SECE authors hold these affiliations, since with fewer it would in practice be one or two people’s paper count. This also covers numbers that could be worked out from others, on this page or with the values of other pages, such as the difference between the two bases or a total minus its shown parts: such values are left out or described in words, the bars concerned have no hatched end (Figure 2, Figure 3) or no split (Figure 1), and Table 1 shows a dash instead of the external-partner share.

  • Matching coverage. Partner institutions require a match to ROR. Of the affiliation occurrences, 94.1% were matched overall and 89.1% outside AUTH (86.7% of the distinct strings outside AUTH); 188 of the 3,086 national or international papers have no matched partner institution at all. Institution counts are therefore lower bounds. Coverage outside AUTH is similar in 2001–2005 (86.7%) and 2021–2025 (87.3%) and lower before 2001 (72.1% in 1996–2000), so comparisons of partner institutions between 2001–2005 and 2021–2025 are not driven by a change in overall matching coverage. Companies are the exception: ROR lists few of them, and 10 of the 15 papers with a company partner in 2001–2005 had a company listed in ROR, but only 42 of the 148 in 2021–2025 (Figure 5). The collaboration types depend much less on the matching, because they also use the country and the institution named in each affiliation. The accuracy of the matching is reported in Methodology & data.

  • Institutions as ROR records them. ROR v2.15 lists CERTH’s Information Technologies Institute without a parent, so ITI (265 papers) and the rest of CERTH (111) are counted as two institutions here and on the other pages. Former TEIs follow ROR too: some are records of their own (such as the Alexander TEI of Thessaloniki), others are recorded as predecessors of the university that absorbed them (the TEI of Western Macedonia counts as the University of Western Macedonia).

  • Sectors are primary ROR types, which can be coarse: national research organisations, for example, are typed government or nonprofit depending on the record. Partners of type archive or other (on 23 papers) are not shown in Figure 4.

  • Companies. The legal-form rule adds companies that ROR lacks but misses companies named without a legal form, so industry co-authorship is probably still understated. The site-wide industry flag (used in Citation impact) counts only companies listed in ROR.

  • Counts are full counts: a paper counts once for every partner institution, sector and type it involves. 2026 is partial (records to 2 November 2026) and is left out of periods, of the shares by year and period and of averages; totals over all years include it.
