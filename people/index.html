def join_and(items: list[str]) -> str:
    """'a', 'a and b', 'a, b and c'."""
    return items[0] if len(items) == 1 else ", ".join(items[:-1]) + " and " + items[-1]

def cohort_ids(lo: int, hi: int) -> pd.Index:
    """Profiles whose first DECE paper appeared in the years lo..hi."""

def returned(ids: pd.Index, lo: int, hi: int) -> float:
    """Share of the profiles `ids` with a DECE paper lo..hi years after their first one."""
    hit = active["author_id"].isin(ids) & active["since"].between(lo, hi)
    return active.loc[hit, "author_id"].nunique() / len(ids)

def any_later(lo: int, hi: int, until: int = LAST_FULL) -> float:
    """Share of a cohort with any DECE paper five or more years after the first (up to `until`)."""
    ids = cohort_ids(lo, hi)
    hit = active["author_id"].isin(ids) & active["since"].ge(FOLLOW) & active["year"].le(until)
    return active.loc[hit, "author_id"].nunique() / len(ids)

def top_tenth(rows: pd.DataFrame, papers: pd.DataFrame) -> dict:
    """The most prolific tenth of the profiles in `rows` (at or above the 90th percentile of DECE
    papers, ties included): their number, the threshold, and the shares of `papers` they appear on
    and of the DECE authorships in `rows` they hold."""
    counts = rows.groupby("author_id")["eid"].nunique()
    threshold = math.ceil(counts.quantile(0.9))
    top = counts.index[counts >= threshold]
    return {"n_top": len(top), "threshold": threshold,
            "paper_share": rows.loc[rows["author_id"].isin(top), "eid"].nunique() / len(papers),
            "authorship_share": counts[top].sum() / counts.sum()}


top_all = top_tenth(dece, pubs)
top_early = top_tenth(dece_base, full[full["period"] == P_BASE])
top_recent = top_tenth(dece_recent, full[full["period"] == P_RECENT])


def concentration_of(rows: pd.DataFrame) -> dict:
    """Measures that do not grow with the number of profiles alone: the Gini coefficient of DECE
    papers per profile (all profiles, and profiles with two or more papers), the share of the DECE
    authorships held by the 20 most prolific profiles together, the DECE authors per paper and the
    share of the profiles with a single DECE paper in `rows` (occasional contributors)."""
    counts = rows.groupby("author_id")["eid"].nunique()
    return {"gini": metrics.gini(counts), "gini2": metrics.gini(counts[counts >= 2]),
            "top20": counts.nlargest(20).sum() / counts.sum(), "per_paper": rows.groupby("eid").size().mean(),
            "single": (counts == 1).mean()}

This page looks at the people behind the department’s publications as a community: how many researchers publish with a DECE affiliation, how the community renews and how long newcomers stay, how concentrated output is, how often DECE researchers hold the first, last and corresponding author positions, and how they write papers together. Researchers are counted as Scopus author profiles, and an author counts as DECE on a paper when their own affiliation on that paper names the department (see Notes & caveats). Key findings

  • An average of 185 author profiles a year published with a DECE affiliation in 2021–2025, 2.1 times as many as in 2001–2005 (88). Output grew faster: 1,283 papers in 2021–2025 against 479 in 2001–2005 (2.7 times as many).

  • The community renews quickly, but few newcomers stay: of the 762 profiles whose first DECE paper appeared in 2001–2020, 48.6% published with a DECE affiliation again within five years and 20.9% in the fifth year after the first. The share coming back within five years fell from 56.5% for the newcomers of 2001–2005 to 40.3% for those of 2016–2020.

  • Output is concentrated: 50.5% of all profiles have a single DECE paper, while the most prolific tenth of the profiles active in 2021–2025 (46 profiles) appear on 84.6% of the department’s papers of those years.

  • A DECE author is first author on 56.1% of the papers of 2021–2025, down from 75.4% in 2001–2005. Part of the decline reflects the rise of international papers (from 21.7% to 50.6% of output), on which DECE authors are first less often: with the 2001–2005 rate of each collaboration type, the 2021–2025 mix alone gives 65.9%.

  • Teams grew mostly through co-authors without a DECE affiliation: their number per paper rose from 1.2 in 2001–2005 to 2.8 in 2021–2025, while the number of DECE-affiliated authors per paper rose only from 2.1 to 2.4.

Size of the research community¶

An author profile is active in a year when it has at least one paper with a DECE affiliation in that year. Figure 1 splits each year’s active profiles by the time since the profile’s first DECE paper; newcomers are profiles whose first DECE paper appeared in that year.

The number of active profiles rose from 70 in 2001 to 210 in 2025, the latest full year (Figure 1), and from an average of 88 a year in 2001–2005 to 185 in 2021–2025. By the Gini coefficient of DECE papers per profile (0.58 in 2001–2005, 0.71 in 2021–2025; 0.47 and 0.61 among profiles with at least two papers) and by the top decile, output became more concentrated: the most prolific tenth of the profiles active in 2021–2025 (46 profiles with at least 15 DECE papers in those years) appear on 84.6% of the papers of 2021–2025 and hold 61.6% of the period’s DECE authorships, against 77.5% and 46.9% for the most prolific tenth of 2001–2005 (23 profiles). The 20 most prolific profiles of each period, however, hold a similar share of its DECE authorships (43.1% in 2001–2005, 44.5% in 2021–2025), so the change reflects a larger number of occasional contributors rather than more output in fewer hands. The share of papers with a top-decile author also rises as papers carry more DECE authors (2.06 on average in 2001–2005, 2.39 in 2021–2025). Concentration of this kind is expected where research groups form around permanent staff while many students and visitors contribute to a few papers each; it says nothing about individual merit.

Notes & caveats¶

  • Profiles are not people. All counts are of Scopus author profiles. Scopus sometimes splits one researcher’s papers over two or more profiles or merges namesakes into one; splits inflate the numbers of profiles, newcomers and one-paper profiles and lower the retention shares.

  • Who counts as DECE. An author counts as DECE on a paper when their own affiliation on that paper has a word starting with “elec” and one starting with “aristot” in the same affiliation string, or an “elec…” department string without an institution name next to an Aristotle University string of the same author (see Methodology & data). Authors who give only “Aristotle University of Thessaloniki”, or only another institution, are not counted, so one person can be DECE on some papers and not on others: 21.8% of the non-DECE authorships of 2021–2025 belong to profiles that carry a DECE affiliation on other papers (for example earlier or later members, or members who listed only another institution on that paper). In 2 records no author is identified as DECE.

  • Profiles are not staff. Profiles include PhD students, post-docs, visitors and external collaborators who listed a DECE affiliation, so the counts are not staff numbers, and the distribution of output is not a measure of individual performance. The page deliberately shows no names, identifiers or per-person figures.

  • Newcomers and retention. A newcomer is a profile with its first DECE paper in this record, which is not the start of employment: the newcomers of the first years include established staff, and people may have published earlier without a DECE affiliation. A profile counts as present in a year only if it has a DECE paper in that year: people may stay without publishing, or leave and keep publishing elsewhere. Counting any DECE paper five or more years after the first (up to 2025) gives 37.7% for the newcomers of 2001–2005 but 24.3% for those of 2016–2020, mainly because the older cohort had more years in which to return; Figure 3 therefore uses the same five-year follow-up for every cohort.

  • Recent profiles. Profiles whose first DECE paper is recent have had little time to add more, so the share of single-paper profiles (50.5%) is partly a matter of timing: among the profiles whose first DECE paper appeared in 2020 or earlier it is 44.3%.

  • Author positions. First, last and corresponding positions are imperfect signals of leadership (alphabetical order and other conventions exist). Scopus has no usable corresponding address for 39.5% of the papers of 1981–2025, and an address counts as DECE only when it names the department.

  • Partial year. 2026 (data to 2 November 2026) is included in the counts over the whole record, 1981–2026 (records, profiles, DECE authorships and each profile’s number of DECE papers, as in Figure 4), in the shares drawn from them, such as the single-paper shares, and in the yearly count of active profiles (Figure 1). It is left out of the other yearly series, of the 5-year periods and newcomer cohorts, and of all comparisons over time.
