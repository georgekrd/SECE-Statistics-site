def join_and(items) -> str:
    """'a', 'a and b', 'a, b and c'."""
    items = [str(i) for i in items]
    return " and ".join(items) if len(items) <= 2 else ", ".join(items[:-1]) + " and " + items[-1]

def fold(text: object) -> str:
    """Lower-case ASCII text without accents ('' for missing values)."""
    if not isinstance(text, str):
        return ""
    decomposed = unicodedata.normalize("NFKD", text)
    return "".join(ch for ch in decomposed if not unicodedata.combining(ch)).lower()


def title_key(text: object) -> str:
    """Matching key of a title: folded, '&' -> 'and', punctuation removed."""
    key = re.sub(r"[^0-9a-z]+", " ", fold(text).replace("&", " and "))
    return re.sub(r"\s+", " ", key).strip()

def publisher_group(name: object) -> str | None:
    """Publisher group of a raw Publisher value (the raw name for small publishers)."""
    if not isinstance(name, str) or not name.strip():
        return None
    low = fold(name)
    return next((group for group, rx in PUBLISHER_RX if rx.search(low)), name.strip())


def name_group(*names: object) -> str | None:
    """Publisher implied by a source title or conference name, if any."""
    for name in names:
        low = fold(name)
        for group, rx in NAME_RX:
            if low and rx.search(low):
                return group
    return None

def normalize_title(title: str) -> str:
    """NFKC, plain hyphens, no bracketed translations, lower case."""
    text = unicodedata.normalize("NFKC", title).translate(DASHES)
    return BRACKETS.sub(" ", text).replace("’", "'").lower()


def singular(word: str) -> str:
    """Conservative English singular (networks -> network, technologies -> technology)."""
    if word in IRREGULAR:
        return IRREGULAR[word]
    if len(word) <= 3 or word in KEEP_S or not word.endswith("s") or word.endswith(("ss", "us", "is", "ics")):
        return word
    if word.endswith("ies") and len(word) > 4:
        return word[:-3] + "y"
    if word.endswith(("sses", "ches", "shes", "xes")):
        return word[:-2]
    return word[:-1]


def tokenize(title: str) -> tuple[list[str], list[str]]:
    """(normalised tokens, surface forms) of a title; hyphenated words are split."""
    text = normalize_title(title)
    for rx, tok in PHRASES:
        text = rx.sub(tok, text)
    raw = [m.group(0).replace("'s", "").replace("'", "") for m in WORD.finditer(text)]
    return [singular(w) for w in raw], raw


def is_content(word: str) -> bool:
    return len(word) > 1 and word not in STOP and not re.fullmatch(r"\d+(?:\.\d+)?", word)


def title_terms(title: str) -> set[str]:
    """Unigrams (content words that are not generic modifiers) and bigrams of adjacent content words."""
    toks, _ = tokenize(title)
    keep = [is_content(w) for w in toks]
    uni = {w for w, k in zip(toks, keep) if k and w not in WEAK}
    bi = {f"{a} {b}" for a, ka, b, kb in zip(toks, keep, toks[1:], keep[1:]) if ka and kb and a != b}
    return uni | bi

def fold_text(ratio: float) -> str:
    """A growth by a factor of about two to four in words, as on Collaboration: 'doubled', 'more than doubled',
    'nearly tripled'..."""
    words = {2: "doubled", 3: "tripled", 4: "quadrupled"}
    assert 1.9 <= ratio < 4.1, f"text: no wording for a growth by {ratio:.2f} times"
    k = round(ratio)
    if abs(ratio - k) <= 0.1:
        return words[k]
    if ratio < k <= ratio + 0.25:
        return f"nearly {words[k]}"
    return f"more than {words[int(ratio)]}"

Output and international co-authorship¶

The School published a record 297 papers in 2025, the latest full year, and 56.9% of them were international, after a high of 57.2% in 2024; the external-partner share, 53.2%, was the highest of any full year (Figure 1).

Highlights¶

  • Research output. The 1,283 publications of 2021–2025 are 16.1% more than in 2016–2020, and teams have grown: a publication had 3.2 authors on average in 2001–2005 and 5.2 in 2021–2025.

  • Citation impact. Citations are concentrated: the most-cited 10% of the papers hold 61.7% of all citations (a median 57.8% within each publication year from 2000 to 2023). Compared with the School’s papers of the same year and type (2000–2023), internationally co-authored papers are cited 21% above the average, mainly those led by partners (a mean normalised score of 1.56 when SECE holds neither the first nor the last author position, 1.03 when it holds one of them), and papers that acknowledge funding 16% above. These are associations, not measured effects.

  • Venues & publishers. IEEE published 54.0% of all records (2,823) and 60.2% of those of 2021–2025. MDPI grew from 3.4% of the output of 2016–2020 to 8.2% in 2021–2025.

  • Research topics. Of the 12 research topics found in the titles, Machine learning gained the most share, from 3.1% of the titles in 2011–2015 to 13.0% in 2021–2025. Fading channels & diversity (8.6% to 2.0%), Electricity markets (9.2% to 3.4%) and Computational EM & photonics (16.3% to 11.2%) lost the most.

  • Collaboration. International co-authorship more than doubled, from 21.7% of the papers in 2001–2005 to 50.6% in 2021–2025, or from 20.0% to 46.7% counting only affiliations of co-authors who are not SECE authors: 50 of the 649 international papers of 2021–2025 are international only through SECE authors’ own second affiliations abroad. Papers with AUTH as the only institution fell from 50.5% to 26.3%.

  • Global reach. 1,953 of the 5,226 papers (37.4%) list at least one affiliation abroad; the most frequent partner country is the United Kingdom (520 papers), ahead of the United States (299). The reach has widened from 11.0 partner countries a year in 2001–2005 to 36.6 in 2021–2025.

  • Funding & open access. The share of publications that acknowledge funding rose from 18.2% in 2011–2015 to 66.0% in 2021–2025 (part of the rise may reflect more complete indexing of acknowledgements by Scopus), and the open-access share from 12.5% to 42.5%. In 2021–2025, journal publications were open access far more often (61.6%) than conference papers (18.5%).

  • Research community. An average of 185 author profiles a year published with an SECE affiliation in 2021–2025, 2.1 times as many as in 2001–2005 (88), while the number of papers with an SECE author grew 2.7-fold.

About this site

This is an unofficial analysis: it is not a publication of AUTH or of the School, and its figures are not official statistics. Every page uses the same Scopus export of 2 November 2026, with the citation counts of that date (self-citations included). Only documents indexed in Scopus are counted, each publication counts once however many co-authors it has (unless a page says otherwise), and the pages show School-level aggregates only: no author names, identifiers or individual metrics. Collaboration figures count every affiliation on a paper, SECE authors’ own second affiliations included, unless they are said to count only affiliations of co-authors who are not SECE authors. Small groups are not shown: a number of papers that link to something (any country abroad, a country, a region, an institution, a sector or a collaboration type) only through SECE authors’ own second affiliations, in any year or period, is given only when it is zero or at least 3 SECE authors hold these affiliations, since with fewer it would in practice be one or two people’s paper count. This also covers numbers that follow from others, on one page or across pages, such as the difference between the two bases or a period minus its years, so values that would reveal such a number are left out (the gaps in the dashed line of Figure 1). Partner institutions come from linking the affiliations to the Research Organization Registry (ROR, v2.15 of 20 October 2026), which succeeded for 89.1% of the affiliation occurrences outside AUTH, on which the partner counts rest (94.1% of all occurrences). Methodology & data describes the sources, the definitions, the validation of this matching and the limitations.
