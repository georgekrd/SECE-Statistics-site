def fmt_date(value) -> str:
    """ISO date text (or a date) -> day without a leading zero, month name and year."""
    ts = pd.Timestamp(value)
    return f"{ts.day} {ts:%B %Y}"


def ci(bounds) -> str:
    """Interval of two ratios as text: (0.9741, 0.9992) -> '97.4–99.9%'."""
    lo, hi = bounds
    return f"{dp.fmt_pct(lo)[:-1]}–{dp.fmt_pct(hi)}"


def fmt_share(value: float) -> str:
    """dp.fmt_pct that never shows '100.0%' for a share below 1 (nor '0.0%' above 0)."""
    text = dp.fmt_pct(value)
    if text == "100.0%" and value < 1:
        return ">99.9%"
    if text == "0.0%" and value > 0:
        return "<0.1%"
    return text


def n_of(n: int, noun: str) -> str:
    """'1 string', '13 strings'."""
    return f"{dp.fmt_int(n)} {noun if n == 1 else noun + 's'}"


def join_and(items) -> str:
    """['a', 'b', 'c'] -> 'a, b and c'."""
    items = [str(i) for i in items]
    return items[0] if len(items) == 1 else ", ".join(items[:-1]) + " and " + items[-1]

# Rates and 95% Wilson intervals are recomputed from the verdict counts in summary.json
# (its stored rates are rounded to four decimals; recomputing avoids double rounding).
Z95 = NormalDist().inv_cdf(0.975)


def wilson(k: int, n: int, z: float = Z95) -> tuple[float, float]:
    """Wilson score interval for k successes in n trials."""
    p = k / n
    centre = (p + z * z / (2 * n)) / (1 + z * z / n)
    half = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / (1 + z * z / n)
    return centre - half, centre + half


def rates(counts: dict, weighted: dict | None = None, judged_n: int | None = None) -> dict:
    """Precision = correct / (correct + wrong); recall = correct / (correct + wrong + missed);
    Wilson intervals on the per-string counts; paper-weighted variants from `weighted`."""
    k, w, m = (int(counts.get(v, 0)) for v in ("correct", "wrong", "missed"))
    out = {**{v: int(counts.get(v, 0)) for v in
              ("correct", "wrong", "missed", "correct_unmatched", "uncertain")},
           "judged": judged_n, "n_p": k + w, "n_r": k + w + m,
           "precision": k / (k + w), "precision_ci": wilson(k, k + w),
           "recall": k / (k + w + m), "recall_ci": wilson(k, k + w + m)}
    if weighted is not None:
        kw, ww, mw = (weighted.get(v, 0) for v in ("correct", "wrong", "missed"))
        out.update(precision_w=kw / (kw + ww), recall_w=kw / (kw + ww + mw), missed_papers=int(mw))
    return out


def summary_rates(entry: dict) -> dict:
    return rates(entry["counts"], entry["weighted_counts"], entry["judged"])

Key findings

  • Scopus applies the search query across the affiliations of a record: 1,800 of the 7,028 records it returned (25.6%) hold the query words only in different affiliations and are left out. The records kept are precise: 5,219 of 5,226 (99.87%) carry an affiliation of the department itself. Of the other 7, 2 give only a country as their affiliation and 5 name AUTH only through other units of it, such as the Department of Physics.

  • Leaving aside society memberships, 94.1% of the affiliation occurrences and 89.1% of those outside AUTH are linked to an organisation in the Research Organization Registry (ROR). Outside AUTH the share drops to 68.0% in records published before 2000.

  • On a fresh random sample of 240 affiliation strings, the matcher’s precision is 99.5% (95% CI 97.5–99.9%) and its recall 95.2% (91.6–97.3%) with ROR v2.15. The legacy matcher it replaces scored 84.1% and 81.5% on the same strings.

  • Several fields are less complete in older records. The publisher is recorded for 29.1% of the records published before 2014 and for 99.0% of those from 2014 to 2025. Funding acknowledgements appear on 18.2% of the 2011–2015 records and on 66.0% of the 2021–2025 records, a rise that may reflect more funded research as well as more complete indexing.

  • Of the 13,025 affiliation occurrences (society memberships aside), 534 (4.1%, on 475 records) are affiliations outside AUTH held only by DECE authors, that is, their own second affiliations; 325 of them are abroad. Counting only affiliations of co-authors who are not DECE authors, 1,856 records are international instead of 1,953, and 46.7% instead of 50.6% of those of 2021–2025.

Data sources¶

The analysis combines three sources (Table 1). The Scopus records come from the advanced-search query AFFIL((elec OR elect OR electr OR electrical) AND (aristotle OR aristotelian)), run on 2 November 2026. It asks for documents with an affiliation that contains one of the words elec, elect, electr or electrical and one of aristotle or aristotelian, but Scopus applies it across the affiliations of a record, so that a physics paper of AUTH with a co-author in an electrical engineering department elsewhere matches too. The pipeline therefore selects the records itself (select_records in dece_stats.pipeline): of the 7,028 records returned, 1,800 were left out because the query words occur only in different affiliations, 1 because it has no affiliations and 1 because it is dated after 2026. The other 5,226 are those with the query words in one affiliation and 2 records whose affiliations show none of the words, which Scopus matched through affiliation data that the export does not show. 2026 is covered only up to the export date (246 records).

From each Scopus record the pipeline reads the EID, title, year, source title, document type, publication stage, citation count (Cited by), DOI, ISSN, publisher, language, conference name, open-access field, affiliations, authors with affiliations, author ids, funding details and correspondence address. Author names are read only to split the per-author affiliation field and to keep names out of the affiliation and funder entries; they are not stored. E-mail addresses are discarded, and the correspondence address is reduced to one flag per record: whether a corresponding author gives a DECE affiliation. No author name, e-mail address or author id appears on this site, and no metric is shown for any individual. The pipeline does not keep the abstracts and keywords of the export, so Research topics relies on titles alone.

Scopus content is subject to Elsevier’s terms of use. The raw export and the per-paper tables stay in the project’s private repository; the site shows aggregates and the bibliographic details of individual highly cited papers.

Processing pipeline¶

The Python package dece_stats turns the inputs into the tables that every page reads (Figure 1).

  1. Load and select (dece_stats.scopus, dece_stats.pipeline): read the export as text, drop duplicate EIDs and select the records (see Data sources). Text is cleaned of HTML entities and e-mail addresses as it is parsed.

  2. Parse: split each record’s affiliation field into distinct affiliation strings, the per-author affiliation field into one chunk per author (linked to the author ids by position), the funding field into funder entries and the open-access field into routes; flag DECE affiliations and DECE authors, and find the authors who hold each affiliation string (see Definitions).

  3. Match (dece_stats.matching, dece_stats.ror): match every distinct affiliation string to ROR and roll the match up to its top-level organisation (see Affiliation matching).

  4. Derive (dece_stats.pipeline): per record, the countries, partner organisations, collaboration type, industry partners and the DECE first-, last- and corresponding-author flags. The country and partner fields are derived twice: from all affiliations (address-based) and without the affiliations outside AUTH that only DECE authors hold (external-partner).

  5. Write the tables listed in Table 2, and metadata.json (versions, dates, record counts, matching and holder summaries), to data/processed, after checking every table again for e-mail addresses. A rerun on the same inputs rewrites identical tables (only the timestamp in metadata.json changes); the tables used here were generated on 2 November 2026.

Affiliation matching¶

Partner, sector and country statistics depend on knowing which institution each affiliation names. Scopus gives affiliations as free text, written differently from paper to paper, so the 7,344 distinct affiliation strings of the data set (13,180 occurrences in records) were matched to ROR by a rule-based matcher that prefers leaving a string unmatched to matching it wrongly:

  1. Country first. The country is parsed from the end of the string (with aliases such as USA, UK, Korea or Türkiye; Hong Kong, Macau and Taiwan are kept separate), independently of the organisation. The matched organisation’s country is used instead only when the string names no country, or when the places in the string point to the organisation’s country.

  2. Parts. The string is split at commas. Postcodes, street addresses, place names and the country are set aside, and the remaining parts are ranked by how much they look like an organisation: universities first, then institutes, hospitals and companies, and department or laboratory names last.

  3. Exact match of a normalised part (accents, abbreviations, word order, organisation words in other languages) against every ROR name, alias and label, preferring organisations in the parsed country.

  4. Acronym match, only within the parsed country and only when the acronym is unique there.

  5. Fuzzy match within the parsed country, requiring a shared distinctive word: a rapidfuzz token-sort similarity of at least 89 (94 for parts without an organisation word, 95 for strings whose country could not be parsed), or a part that differs from a single organisation’s name only by place names or generic words. Department-level parts are never fuzzy-matched, and fuzzy or acronym candidates that are universities or hospitals far from every place named in the string are rejected.

  6. Ignore and override. Strings that only state a society membership (such as IEEE or the Technical Chamber of Greece) are ignored. Manual overrides, applied last, set the match of 17 strings.

  7. Roll-up through ROR’s parent links to the top-level organisation, so that a research institute counts with its parent organisation. Universities stay their own top level and are not merged into university systems or federations.

Validation of the matching¶

The matcher was checked on random samples of distinct affiliation strings, in two rounds of 240 strings each. For every string an AI assistant (Claude agents) identified the true organisation, looked it up in the ROR dump then in use, v1.66 (names, aliases, acronyms, relations and status), and compared it with the top-level organisation returned by each matcher. Round 1 measured a first version of the new matcher, and its errors were used to improve it, so it is not an independent test. Round 2 is a fresh sample drawn after those fixes and gives the headline estimate. The legacy matcher, which fuzzy-matched every comma-separated part of a string against the ROR names of all countries and which the new one replaces, was judged on both samples.

With ROR v2.15 of 20 October 2026, the final matcher gives a different result for 6 round-2 strings: it now matches 5 that it had left unmatched and returns another organisation for 1. The changed strings were judged again against the new dump, and the strings judged missed or without a ROR record were checked for records added or removed since (25 strings in all, listed in data/validation/round2_recheck.csv): all 6 of the changed strings are now correct (3 had been missed, 2 name organisations that ROR v1.66 did not list and 1 was correct before and now goes to the organisation's ROR successor), and the verdicts of the other 19 re-checked strings stand. The round-2 figures of the final matcher below are those with the current dump.

Each string received one verdict per matcher, judged at the level of the top-level organisation:

  • correct: the right organisation, or a ROR parent or successor of it;

  • wrong: a different organisation;

  • missed: no match, although ROR has a suitable record;

  • no ROR record: no match, and ROR has no suitable record (this includes strings that name no organisation);

  • uncertain: the judge could not decide; left out of the metrics.

Precision = correct / (correct + wrong) is how often a returned organisation is right. Recall = correct / (correct + wrong + missed) is the share of matchable strings that end up with the right organisation. The 95% confidence intervals are Wilson score intervals. The paper-weighted variants weight each string by the number of papers in which it occurred when its sample was drawn.

Definitions¶

Publication, paper and record
One Scopus record of any document type; the pages use the three words interchangeably.
DECE affiliation
An affiliation string with a word starting with elec and a word starting with aristot, ignoring case and accents. This mirrors the search query; 1,766 of the 7,344 distinct strings qualify.
DECE author
An author whose own affiliation text contains a DECE affiliation string. Scopus sometimes splits one affiliation into an AUTH string and a department string, so an elec… unit that names no institution, listed next to an aristot… string by the same author, also counts. 11,762 of the 22,180 author positions are DECE authors, and every record with a DECE affiliation has at least one. The DECE first-, last- and corresponding-author shares build on these flags; for the corresponding author, the affiliations in the correspondence address are tested with the same rule.
AUTH affiliation
A string whose top-level ROR organisation is AUTH or, when it is unmatched, that names Aristotle University (or AUTh).
Holder of an affiliation
An author of the record whose own affiliation text (the Scopus field Authors with affiliations) contains the affiliation string. Holders are found with the same decomposition of the author’s text as the DECE-author flag: a string that sits inside a longer string of the same author is held by that author too, and an author whose text is empty holds nothing. An affiliation can have several holders, some of them DECE authors and some not.
Dual affiliation and DECE-only affiliation
An author with a dual affiliation lists two or more affiliations on the same paper, such as a DECE author who also holds a joint, adjunct or visiting position at another institution. Scopus lists all of them among the record’s affiliations, next to those of the co-authors. A DECE-only affiliation is an affiliation outside AUTH whose holders (one or more) are all DECE authors: a DECE author’s own second affiliation, not the affiliation of a co-author based there. As soon as one holder is not a DECE author, the affiliation is not DECE-only.
Partner affiliation and partner institution
A partner affiliation is an affiliation outside AUTH that is matched to ROR, or that has a country and names an institution; bare place names and strings that list only departments or laboratories (usually units of AUTH itself) do not count. Partner institutions are the distinct top-level ROR organisations other than AUTH: 810 in the data, 805 of them named by at least one affiliation that is not DECE-only (the other 5 appear only as DECE authors’ own second affiliations). A partner institution counts as abroad when an affiliation located outside Greece names it.
Collaboration type
International when at least one affiliation is in a country other than Greece, whether or not it is matched to ROR (AUTH strings count as Greece); otherwise national when the record has a partner affiliation; otherwise single-institution, that is AUTH only, possibly several of its units. The split is 2,140 single-institution, 1,133 national and 1,953 international records. For 83 of the international records, the only affiliations abroad are unmatched strings whose country was read from the text. These are the address-based types (next entry).
Address-based and external-partner indicators
The collaboration type and the other partner and country indicators (international co-authorship, partner institutions and countries, national and industry partners) are address-based: every affiliation of a record counts, including a DECE author’s own second affiliation. This is the standard definition (SciVal and the Leiden Ranking also count international collaboration from the countries in the affiliation addresses), and it gives the headline figures on every page. The external-partner variant counts only affiliations of co-authors who are not DECE authors: it leaves out the DECE-only affiliations and keeps the AUTH affiliations and every affiliation with at least one holder who is not a DECE author. An affiliation without any holder also counts as external, since nothing shows that a DECE author holds it. On this basis the split is 2,308 single-institution, 1,062 national and 1,856 international records: 97 records are international only through DECE authors’ own second affiliations (DECE-only affiliations; 65 of them become single-institution, 32 national), and 103 national records become single-institution. In 2021–2025, 50.6% of the records are international on the address basis and 46.7% on the external-partner basis.
Small groups
A number of papers that link to an institution, country, region, sector or collaboration type only through DECE authors’ own second affiliations can rest on one or two DECE authors, and would then in practice be their personal paper count. The small-group rule therefore allows such a number to be shown, or to follow from the figures shown, only when it is zero or at least 3 distinct DECE authors (Scopus author ids) hold these affiliations. The rule applies to totals over all institutions or countries as well as to single years, periods and subgroups, and to numbers that follow from other figures, on the same page or across pages, such as the difference between an address-based value and its external-partner value, a total minus the parts shown, or a share times its denominator. The Overview, Citation impact, Collaboration, Global reach and Funding & open access apply it: where a number would break the rule, they leave it out, together with enough other values that it cannot be recovered by subtraction, or describe it in words. The numbers for all countries together, how many papers change collaboration type between the two bases, are chosen once for the whole site, by year, by period and for a few spans of years such as all years and 2021–2025, and checked all together, taking into account that a value left out stands for at least one paper. On this page every such number, including those that follow from the split above, rests on at least 3 DECE authors, and those for all countries together cover all years or 2021–2025.
Sector and industry partner
The sector of a partner institution is its primary ROR type, taken in the order education, healthcare, company, facility, government, nonprofit, archive, other (funder only when nothing else is listed). A record has an industry partner when a matched partner organisation, or its top-level organisation, has the ROR type company (158 records; 143 on the external-partner basis).
Source type
Journal (Scopus document types article, review, letter, editorial, note, erratum, data paper, short survey), conference (conference paper, conference review), book (book, book chapter) or other. It is derived from the document type, not from the venue, so a conference paper published in a journal issue counts as conference.
Open-access category
From the Scopus Open Access field: gold, hybrid (Scopus “Hybrid Gold”), bronze and green. A record with several routes is counted once, in that order of priority; a record with none is closed. The field gives the status at the export date, not at publication.
Partial year and periods
2026 is covered only up to 2 November 2026 (246 records, 189 of them journal articles: conference proceedings reach Scopus months after the event). Charts of yearly counts show it, marked as partial. Shares, means and ratios by year leave it out, as do trends, growth rates and period comparisons, so 2025 is the latest full year. Totals and all-years shares (as in Table 5) include it, and the data-quality charts on this page show it as a period of its own. Periods are five-year bins that end with the latest full year: recent means 2021–2025, the latest five full years, the period before it is 2016–2020, and so on back to 1981–1985. A newer export moves every period along.
Citations and h-index
Citation counts are the Scopus Cited by values at the export date, self-citations included. The h-index of a set of papers is the largest number h such that h of the papers have at least h citations each.
Normalised citation score
A paper’s citations divided by the mean citations of all records in this data set with the same publication year and type, so that 1 is the department’s own average for comparable papers. Conference papers are compared with conference papers. Articles and reviews are compared together, because there are too few reviews for a yearly average of their own (59 in 2000–2023, at most 13 in a year). It is a within-department baseline, not a field-normalised world baseline such as Scopus’ Field-Weighted Citation Impact, and it averages 1 in every year by construction.

Limitations¶

  • Selection by affiliation text. Records are selected by the query words in their affiliations (Data sources). DECE papers whose affiliation is written without them (for instance only “AUTh”, or the former name “University of Thessaloniki”), or split into separate department and university affiliations, are missing unless another affiliation of the paper contains the query words, and so are papers by staff who gave only another affiliation.

  • Scopus coverage. Only Scopus-indexed documents are included. Coverage of conference proceedings, books and older literature is incomplete, so early annual counts reflect indexing as much as output, and the data set is not a complete publication list of the department.

  • Citations are a snapshot taken on 2 November 2026, include self-citations and count only citing documents indexed in Scopus. Recent papers have had little time to be cited, and research fields cite at different rates.

  • Normalised score. The normalised citation score averages 1 in every year by construction; it cannot show trends in impact or impact relative to the world.

  • No per-capita indicators. No counts are normalised by staff numbers; per-capita productivity would need official staff counts per year, and the author profiles mix staff, students and visitors.

  • Author profiles. Scopus author ids can be split (one person with several ids) or merged. DECE status is inferred from affiliation text, and the profiles include doctoral students, post-docs and visitors as well as faculty.

  • Affiliation matching. 10.9% of the affiliation occurrences outside AUTH are unmatched, more in older records; ROR lacks some organisations and parent links; and the validation rests on samples of 240 strings judged with AI assistance.

  • Affiliation holders. Holders are read from the per-author affiliation text with the same rule as the DECE-author flag and share its limits. An author whose text is empty holds nothing, and a string inside a longer string of the same author counts as held by that author; a check on the raw Scopus export found that such edge cases hardly change the external-partner figures. DECE status is decided paper by paper, so a former DECE member who lists only another institution counts as a co-author who is not a DECE author. A DECE author’s second affiliation can also reflect a real link with that institution, which is why the address-based figures remain the headline and the external-partner figures complement them.

  • Topics are inferred from titles alone.

  • Associations, not causes. Comparisons between groups of papers (for example funded and unfunded papers) describe associations; they do not control for research field, team size or other differences.
